cmake_minimum_required(VERSION 3.28)
project(obs-spatial-fx VERSION 3.4.2 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows x64 with VS2022 MSVC is required")
endif()
add_compile_options(/utf-8 /W4 /permissive- /Zc:__cplusplus /wd4201 /we4996)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
set(OBS_ROOT "$ENV{OBS_ROOT}" CACHE PATH "Read-only OBS32+ installation; tools/build.cmd detects it automatically")
set(OBS_SOURCE "${CMAKE_SOURCE_DIR}/third_party/obs-studio-32.0.0" CACHE PATH "Official OBS32 source headers")
set(SPATIAL_PHASE 4 CACHE STRING "0=official template smoke, 1=dock, 2-4=feature build")
set(OBS_INCLUDE "${OBS_SOURCE}/libobs")
set(OBS_FRONTEND_INCLUDE "${OBS_SOURCE}/frontend/api")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
configure_file(src/obsconfig.h.in generated/obsconfig.h @ONLY)
foreach(name obs obs-frontend-api)
  add_library(${name} SHARED IMPORTED)
  set_target_properties(${name} PROPERTIES IMPORTED_IMPLIB "${CMAKE_SOURCE_DIR}/tools/sdk/${name}.lib" IMPORTED_LOCATION "${OBS_ROOT}/bin/64bit/${name}.dll")
endforeach()
if(SPATIAL_PHASE EQUAL 0)
  set(CMAKE_PROJECT_VERSION "1.0.0")
  configure_file(third_party/obs-plugintemplate/src/plugin-support.c.in generated/plugin-support.c @ONLY)
  add_library(obs-template-smoke MODULE third_party/obs-plugintemplate/src/plugin-main.c "${CMAKE_BINARY_DIR}/generated/plugin-support.c")
  target_include_directories(obs-template-smoke PRIVATE "${OBS_INCLUDE}" "${CMAKE_BINARY_DIR}/generated" third_party/obs-plugintemplate/src)
  target_link_libraries(obs-template-smoke PRIVATE obs)
  return()
endif()
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/third_party/qt6")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
add_library(obs-spatial-fx MODULE src/fx_panel.cpp src/fx_blur.cpp src/fx_aa.cpp src/fx_renderer.cpp src/plugin.cpp src/dock.cpp src/dock_actions.cpp src/dock_v2.cpp src/dock_v3.cpp src/widgets.cpp src/editor.cpp src/editor_timeline.cpp src/editor_curves.cpp src/editor_curve_widget.cpp src/editor_help.cpp src/editor_audio.cpp src/audio/beat_analyzer.cpp src/audio/audio_decoder_win.cpp)
target_include_directories(obs-spatial-fx PRIVATE "${OBS_INCLUDE}" "${OBS_FRONTEND_INCLUDE}" "${CMAKE_BINARY_DIR}/generated" src)
target_compile_definitions(obs-spatial-fx PRIVATE SPATIAL_PHASE=${SPATIAL_PHASE})
target_link_libraries(obs-spatial-fx PRIVATE obs obs-frontend-api Qt6::Core Qt6::Gui Qt6::Widgets mfplat mfreadwrite mfuuid ole32 propsys)
install(TARGETS obs-spatial-fx RUNTIME DESTINATION bin/64bit LIBRARY DESTINATION bin/64bit)
install(DIRECTORY data/ DESTINATION data)
if(SPATIAL_PHASE GREATER_EQUAL 2)
  target_sources(obs-spatial-fx PRIVATE src/core.cpp src/model.cpp src/scene_evaluator.cpp src/stage.cpp)
  enable_testing()
  add_executable(spatial-core-tests tests/core_tests.cpp src/core.cpp)
  target_include_directories(spatial-core-tests PRIVATE src)
  add_test(NAME spatial-core COMMAND spatial-core-tests)
endif()

option(SPATIAL_BUILD_QA_DRIVER "Build isolated OBS integration driver; never packaged" ON)
if(SPATIAL_BUILD_QA_DRIVER)
  add_executable(spatial-capture-fixture WIN32 tests/window_fixture.cpp)
  target_link_libraries(spatial-capture-fixture PRIVATE user32 gdi32)
  add_library(spatial-qa-driver MODULE tests/obs_qa_driver.cpp)
  target_include_directories(spatial-qa-driver PRIVATE "${OBS_INCLUDE}" "${OBS_FRONTEND_INCLUDE}" "${CMAKE_BINARY_DIR}/generated")
  target_link_libraries(spatial-qa-driver PRIVATE obs obs-frontend-api Qt6::Core Qt6::Gui Qt6::Widgets)
endif()
if(SPATIAL_PHASE GREATER_EQUAL 2)
  add_executable(spatial-model-tests tests/model_tests.cpp src/model.cpp src/core.cpp src/scene_evaluator.cpp)
  target_include_directories(spatial-model-tests PRIVATE src "${OBS_INCLUDE}" "${CMAKE_BINARY_DIR}/generated")
  target_link_libraries(spatial-model-tests PRIVATE obs)
  add_test(NAME spatial-model COMMAND spatial-model-tests)
  set_tests_properties(spatial-model PROPERTIES ENVIRONMENT "PATH=${OBS_ROOT}/bin/64bit\;$ENV{PATH}")
endif()


if(SPATIAL_PHASE GREATER_EQUAL 4)
  file(GLOB FX_SHADER_ASSETS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/data/effects/*.body")
  configure_file(installer/setup.rc.in generated/setup.rc @ONLY)
  set_source_files_properties("${CMAKE_BINARY_DIR}/generated/setup.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_BINARY_DIR}/obs-spatial-fx.dll;${CMAKE_SOURCE_DIR}/README.md;${CMAKE_SOURCE_DIR}/data/effects/spatial.effect;${FX_SHADER_ASSETS};${CMAKE_SOURCE_DIR}/data/effects/AA_NOTICES.txt")
  add_executable(spatial-setup WIN32 installer/setup.cpp "${CMAKE_BINARY_DIR}/generated/setup.rc")
  add_dependencies(spatial-setup obs-spatial-fx)
  set_property(TARGET spatial-setup PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_target_properties(spatial-setup PROPERTIES OUTPUT_NAME OBS_Spatial_FX_V3.4.2_Setup)
  target_link_libraries(spatial-setup PRIVATE shell32 ole32 advapi32 version uuid)
  target_link_options(spatial-setup PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
endif()

 target_link_libraries(${target} PRIVATE spatial-fx-model)
